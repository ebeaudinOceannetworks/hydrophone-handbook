8. ONC Hydrophone Network

Ocean Networks Canada operates a massive spatial array of underwater acoustics monitoring across the country. Each hydrophone has a designated Device Code (ex: ICLISTENHF6016) and is located at a designated Location code (ex: CSQH.H1). The Location Code is used by the database, but they also have a Colloquial Name (ex: Clayoquot Slope). In addition, there are two observatories VENUS and NEPTUNE, which regroup several sites.

Most locations have a single hydrophone, while some sites have hydrophone arrays, which constitutes several hydrophones mounted together on a platform. The hydrophone arrays naming convention has .H1, .H2, etc., added to the location code, to denote the position of the hydrophone within the array.

Main locations

  • Pacific: NEPTUNE (offshore) and VENUS (coastal) observatories.
  • Arctic: Cambridge Bay, Gascoyne Inlet, NU.
  • Atlantic: Conception Bay, Holyrood, NL.

Map of all hydrophone deployments

L = require("leaflet@1.9.4")
html`<link rel="stylesheet" href="https://unpkg.com/leaflet@1.9.4/dist/leaflet.css">`
container = html`<div style="height: 600px; border-radius: 10px; border: 1px solid #ddd;"></div>`
yield container;
{
  const map = L.map(container).setView([58, -95], 3);

  L.tileLayer('https://server.arcgisonline.com/ArcGIS/rest/services/Ocean/World_Ocean_Base/MapServer/tile/{z}/{y}/{x}', {
    attribution: 'Esri & Sources',
    maxZoom: 13
  }).addTo(map);

  const inactiveLayer = L.layerGroup();
  const activeLayer = L.layerGroup().addTo(map);

  if (cable_data) {
    L.geoJSON(cable_data, {
      style: {
        color: 'grey',
        weight: 1,
        opacity: 0.7
      }
    }).addTo(map);
  }

  function dateCell(value) {
    return value ? value : "—";
  }

  function deploymentTable(rows, includeEnd) {
    if (!rows.length) return "";
    const head = includeEnd
      ? "<tr><th>Device</th><th>From</th><th>To</th></tr>"
      : "<tr><th>Device</th><th>From</th></tr>";
    const body = rows.map(r => includeEnd
      ? `<tr><td>${r.device}</td><td>${dateCell(r.begin)}</td><td>${dateCell(r.end)}</td></tr>`
      : `<tr><td>${r.device}</td><td>${dateCell(r.begin)}</td></tr>`
    ).join("");
    return `<table class="hydro-popup-table">${head}${body}</table>`;
  }

  status_data.siteCode.forEach((code, i) => {
    const isActive = status_data.any_active[i];
    const color = isActive ? "#28a745" : "#6c757d";
    const cleanName = status_data.locationName[i].replace(/Hydrophone [A-Z]/g, "Hydrophone Array");

    const deployments = status_data.deviceCode[i].map((dev, idx) => ({
      device: dev,
      location: status_data.locationCode[i][idx],
      begin: status_data.begin[i][idx],
      end: status_data.end[i][idx]
    }));

    const activeRows = deployments.filter(d => !d.end);
    const previousRows = deployments.filter(d => d.end).sort((a, b) => (b.end || "").localeCompare(a.end || ""));

    const activeBlock = activeRows.length
      ? `<p class="hydro-popup-heading hydro-popup-current">Current deployments</p>${deploymentTable(activeRows, false)}`
      : `<p class="hydro-popup-heading hydro-popup-previous">No current deployments</p>`;

    const previousBlock = previousRows.length
      ? `<p class="hydro-popup-heading hydro-popup-previous">Previous deployments</p>${deploymentTable(previousRows, true)}`
      : "";

    const depthVal = status_data.depth[i];
    const depth = (depthVal == null || Number.isNaN(Number(depthVal))) ? "—" : `${depthVal} m`;

    const marker = L.circleMarker([status_data.lat[i], status_data.lon[i]], {
      radius: 8,
      fillColor: color,
      color: "white",
      weight: 2,
      fillOpacity: 0.9
    })
    .bindPopup(`
      <div style="font-family: sans-serif; min-width: 360px;">
        <b style="font-size: 1.1em;">${cleanName}</b><br>
        <span style="color: #666; font-size: 0.9em;">${code}</span>
        <hr style="margin: 5px 0;">
        <p style="margin: 5px 0 0 0;"><b>Depth:</b> ${depth}</p>
        ${activeBlock}
        ${previousBlock}
      </div>
    `, { minWidth: 360, maxWidth: 480, maxHeight: 340 });

    if (isActive) {
      marker.addTo(activeLayer);
    } else {
      marker.addTo(inactiveLayer);
    }
  });

  const overlays = {
    "<span style='color: #28a745; font-weight: bold;'>● Current Sites</span>": activeLayer,
    "<span style='color: #6c757d; font-weight: bold;'>○ Inactive Sites</span>": inactiveLayer
  };

  L.control.layers(null, overlays, { collapsed: false }).addTo(map);
}
